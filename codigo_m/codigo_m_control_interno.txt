let
    // 1. CARGA DE ARCHIVO Y MAESTRO DE MÁQUINAS SAP
    RutaBase = "C:\Proyectos Python\Detallados\Estructura base\Rockdrill_Control_Operaciones",
    RutaCI = RutaBase & "\00_Control_Interno\RD.402.P.01.F.04  Consolidado de Avance Julio.xlsx",
    RutaMaestro = RutaBase & "\Maestro_Maquinas\Maestros_Maquinas.xlsx",
    
    MaestroBin = try File.Contents(RutaMaestro) otherwise null,
    MaestroWB = if MaestroBin <> null then Excel.Workbook(MaestroBin, null, true) else null,
    BusquedaHoja = if MaestroWB <> null then Table.SelectRows(MaestroWB, each Text.Contains(Text.Upper([Name]), "EXEP") or Text.Contains(Text.Upper([Name]), "EXCEP")) else null,
    HojaExcepciones = if BusquedaHoja <> null and Table.RowCount(BusquedaHoja) > 0 then BusquedaHoja{0}[Data] else null,
    PromotedExcepciones = if HojaExcepciones <> null then try Table.PromoteHeaders(HojaExcepciones, [PromoteAllScalars=true]) otherwise #table({"Nombre Erroneo", "Nombre Oficial"}, {}) else #table({"Nombre Erroneo", "Nombre Oficial"}, {}),
    
    MaestroLimpiado = if Table.HasColumns(PromotedExcepciones, {"Nombre Erroneo", "Nombre Oficial"}) then 
        Table.TransformColumns(PromotedExcepciones, {
            {"Nombre Erroneo", each Text.Upper(Text.Trim(Text.From(_ ?? ""))), type text},
            {"Nombre Oficial", each Text.Upper(Text.Trim(Text.From(_ ?? ""))), type text}
        })
    else #table({"Nombre Erroneo", "Nombre Oficial"}, {}),

    // 2. LECTURA ADAPTATIVA DE HOJAS DIARIAS (26.06 a 25.07)
    CIBin = File.Contents(RutaCI),
    CIWB = Excel.Workbook(CIBin, null, true),
    HojasDiarias = Table.SelectRows(CIWB, each [Kind] = "Sheet" and Text.Length([Name]) = 5 and Text.Middle([Name], 2, 1) = "."),
    
    ProcesarHojas = Table.AddColumn(HojasDiarias, "TablaExtraida", each 
        let
            hNombre = [Name],
            hData = [Data],
            diaStr = Text.Start(hNombre, 2),
            mesStr = Text.End(hNombre, 2),
            fechaISO = try Date.From("2026-" & mesStr & "-" & diaStr) otherwise null,
            
            Filas9 = Table.Skip(hData, 9),
            PosTotal = List.PositionOf(Table.Column(Filas9, "Column3"), "TOTAL AVANCE", Occurrence.First, (val, criteria) => Text.Contains(Text.Upper(Text.From(val ?? "")), "TOTAL AVANCE")),
            Acotada = if PosTotal <> -1 then Table.FirstN(Filas9, PosTotal) else Filas9,
            
            RenombrarColsRaw = Table.RenameColumns(Acotada, {
                {"Column1", "CTR_RAW"}, {"Column3", "MAQUINA_RAW"}, {"Column5", "SE_PERFORO"}, {"Column7", "METRAJE_CI"}
            }, MissingField.Ignore),
            
            ReplaceCTRBlank = Table.ReplaceValue(RenombrarColsRaw, each [CTR_RAW], each 
                let 
                    txt = Text.Upper(Text.Trim(Text.From([CTR_RAW] ?? "")))
                in 
                    if List.Contains({"CONTRATO", "EQUIPO", "AVANCE", "SISTEMA", "TOTAL", "SDR", ""}, txt) then null else [CTR_RAW], 
                Replacer.ReplaceValue, {"CTR_RAW"}),
                
            FillDownCTR = Table.FillDown(ReplaceCTRBlank, {"CTR_RAW"}),
            
            FiltrarMaquinas = Table.SelectRows(FillDownCTR, each 
                let 
                    mTxt = Text.Upper(Text.Trim(Text.From([MAQUINA_RAW] ?? "")))
                in 
                    mTxt <> "" and mTxt <> "EQUIPO" and mTxt <> "SUB" and mTxt <> "SUP"
            ),
            
            AddFecha = Table.AddColumn(FiltrarMaquinas, "FECHA", each fechaISO, type date)
        in
            AddFecha
    ),
    
    ConsolidadoCI = Table.Combine(ProcesarHojas[TablaExtraida]),
    
    // 3. ESTANDARIZACIÓN DE CTR Y MÁQUINA CON MAESTRO SAP
    CleanCTR = Table.TransformColumns(ConsolidadoCI, {
        {"CTR_RAW", (c) => 
            let
                c1 = Text.Upper(Text.Clean(Text.Trim(Text.From(c ?? "")))),
                c2 = if c1 = "CUCULII" or c1 = "CUCULÍ" then "CUCULI" 
                     else if Text.Contains(c1, "SAN CRIST") then "SAN CRISTOBAL"
                     else if c1 = "LA ESTRELLA" or c1 = "ESTRELLA" then "LA ESTRELLA"
                     else c1
            in
                c2, type text}
    }),
    RenameCTRCol = Table.RenameColumns(CleanCTR, {{"CTR_RAW", "CTR"}}),
    ExcluirColquijirca = Table.SelectRows(RenameCTRCol, each [CTR] <> "COLQUIJIRCA"),
    CleanMaquinaRaw = Table.AddColumn(ExcluirColquijirca, "MAQUINA_TEMP", each Text.Upper(Text.Clean(Text.Trim(Text.From([MAQUINA_RAW] ?? "")))), type text),

    // Cruce con Maestro SAP
    #"Merge SAP" = Table.NestedJoin(CleanMaquinaRaw, {"MAQUINA_TEMP"}, MaestroLimpiado, {"Nombre Erroneo"}, "MatchSAP", JoinKind.LeftOuter),
    #"Expandir SAP" = Table.ExpandTableColumn(#"Merge SAP", "MatchSAP", {"Nombre Oficial"}, {"Nombre Oficial"}),
    #"Reemplazar Maquina SAP" = Table.AddColumn(#"Expandir SAP", "MAQUINA", each if [Nombre Oficial] <> null and [Nombre Oficial] <> "" then [Nombre Oficial] else [MAQUINA_TEMP]),
    #"Limpiar Cols SAP" = Table.RemoveColumns(#"Reemplazar Maquina SAP", {"MAQUINA_TEMP", "Nombre Oficial"}),

    // 4. LÓGICA DE TURNOS HÍBRIDA EN CONTROL INTERNO (1ª aparición -> 'A', 2ª -> 'B')
    AgruparPorMaquina = Table.Group(#"Limpiar Cols SAP", {"FECHA", "MAQUINA"}, {
        {"FilasConTurno", (subTable) => 
            let
                Indexado = Table.AddIndexColumn(subTable, "Secuencia", 1, 1, Int64.Type),
                TurnoStd = Table.AddColumn(Indexado, "TURNO_ESTANDAR", each if [Secuencia] = 1 then "A" else "B", type text)
            in
                TurnoStd, type table}
    }),
    ExpandirTabla = Table.Combine(AgruparPorMaquina[FilasConTurno]),

    // 5. CLAVE ÚNICA ESTRICTA: {FECHA}|{MAQUINA}|{TURNO_ESTANDAR}
    IDClaveAgregado = Table.AddColumn(ExpandirTabla, "ID_CLAVE_UNICA", each 
        (try Date.ToText(Date.From([FECHA]), "yyyy-MM-dd") otherwise "SIN_FECHA") & "|" & [MAQUINA] & "|" & [TURNO_ESTANDAR], type text),

    ColsFinales = {"FECHA", "CTR", "MAQUINA_RAW", "MAQUINA", "SE_PERFORO", "TURNO_ESTANDAR", "METRAJE_CI", "ID_CLAVE_UNICA"},
    FiltrarColsFinales = Table.SelectColumns(IDClaveAgregado, ColsFinales, MissingField.Ignore),

    TypeTexto = Table.TransformColumns(FiltrarColsFinales, {
        {"CTR", (v) => try Text.Trim(Text.From(v)) otherwise null, type text},
        {"MAQUINA_RAW", (v) => try Text.Trim(Text.From(v)) otherwise null, type text},
        {"MAQUINA", (v) => try Text.Trim(Text.From(v)) otherwise null, type text},
        {"SE_PERFORO", (v) => try Text.Trim(Text.From(v)) otherwise null, type text},
        {"TURNO_ESTANDAR", (v) => try Text.Trim(Text.From(v)) otherwise null, type text},
        {"ID_CLAVE_UNICA", (v) => try Text.Trim(Text.From(v)) otherwise null, type text}
    }),
    TypeNum = Table.TransformColumns(TypeTexto, {{"METRAJE_CI", (v) => try Number.From(Text.Replace(Text.Trim(Text.From(v)), ",", ".")) otherwise 0.0, type number}}),
    TypeFecha = Table.TransformColumnTypes(TypeNum, {{"FECHA", type date}})
in
    TypeFecha
