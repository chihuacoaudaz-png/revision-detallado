let
    // =========================================================================
    // 1. FUENTES DE DATOS (REFERENCIA A TUS CONSULTAS EN POWER QUERY)
    // Se conecta a tus consultas 'Detallados_BD' y 'Consolidado_BD'
    // =========================================================================
    OrigenDetallados = #"Detallados_BD",
    OrigenControlInterno = #"Consolidado_BD",

    // =========================================================================
    // 2. AGRUPAR METRAJES POR CLAVE ÚNICA (ID_CLAVE_UNICA)
    // =========================================================================
    // Agrupar Detallados por ID_CLAVE_UNICA
    GroupDetallados = Table.Group(OrigenDetallados, {"ID_CLAVE_UNICA"}, {
        {"FECHA", each List.First(List.RemoveNulls([FECHA])), type date},
        {"CTR", each List.First(List.RemoveNulls([CTR])), type text},
        {"MAQUINA", each List.First(List.RemoveNulls([MAQUINA])), type text},
        {"TURNO_ESTANDAR", each List.First(List.RemoveNulls([TURNO_ESTANDAR])), type text},
        {"SONDAJE", each Text.Combine(List.Distinct(List.Select(List.Transform([SONDAJE], Text.From), each _ <> null and _ <> "")), ", "), type text},
        {"METRAJE_DETALLADO", each Number.Round(List.Sum([METRAJE]), 2), type number}
    }),

    // Agrupar Control Interno por ID_CLAVE_UNICA
    GroupCI = Table.Group(OrigenControlInterno, {"ID_CLAVE_UNICA"}, {
        {"METRAJE_CONTROL_INTERNO", each Number.Round(List.Sum([METRAJE_CI]), 2), type number}
    }),

    // =========================================================================
    // 3. CRUCE FULL OUTER JOIN POR ID_CLAVE_UNICA
    // =========================================================================
    MergeFull = Table.NestedJoin(GroupDetallados, {"ID_CLAVE_UNICA"}, GroupCI, {"ID_CLAVE_UNICA"}, "MatchCI", JoinKind.FullOuter),
    ExpandCI = Table.ExpandTableColumn(MergeFull, "MatchCI", {"ID_CLAVE_UNICA", "METRAJE_CONTROL_INTERNO"}, {"ID_CLAVE_UNICA_CI", "METRAJE_CI_EXP"}),

    // =========================================================================
    // 4. RESOLVER METADATOS Y CÁLCULO DE DIFERENCIA
    // =========================================================================
    AddKeyFinal = Table.AddColumn(ExpandCI, "ID_CLAVE_UNICA_FINAL", each if [ID_CLAVE_UNICA] <> null and [ID_CLAVE_UNICA] <> "" then [ID_CLAVE_UNICA] else [ID_CLAVE_UNICA_CI], type text),
    
    AddPartesKey = Table.AddColumn(AddKeyFinal, "PartesKey", each Text.Split([ID_CLAVE_UNICA_FINAL] ?? "", "|")),
    
    AddFechaFinal = Table.AddColumn(AddPartesKey, "FECHA_FINAL", each if [FECHA] <> null then [FECHA] else try Date.From([PartesKey]{0}) otherwise null, type date),
    AddCTRFinal = Table.AddColumn(AddFechaFinal, "CTR_FINAL", each if [CTR] <> null then [CTR] else try [PartesKey]{1} otherwise "OTRO", type text),
    AddMaqFinal = Table.AddColumn(AddCTRFinal, "MAQUINA_FINAL", each if [MAQUINA] <> null then [MAQUINA] else try [PartesKey]{2} otherwise "DESCONOCIDO", type text),
    AddTurnoFinal = Table.AddColumn(AddMaqFinal, "TURNO_FINAL", each if [TURNO_ESTANDAR] <> null then [TURNO_ESTANDAR] else try [PartesKey]{3} otherwise "A", type text),

    CleanMetDet = Table.AddColumn(AddTurnoFinal, "MET_DET", each if [METRAJE_DETALLADO] <> null then [METRAJE_DETALLADO] else 0.0, type number),
    CleanMetCI = Table.AddColumn(CleanMetDet, "MET_CI", each if [METRAJE_CI_EXP] <> null then [METRAJE_CI_EXP] else 0.0, type number),

    AddDiferencia = Table.AddColumn(CleanMetCI, "DIFERENCIA", each Number.Round([MET_DET] - [MET_CI], 2), type number),
    AddAbsDiferencia = Table.AddColumn(AddDiferencia, "ABS_DIFERENCIA", each Number.Abs([DIFERENCIA]), type number),

    // =========================================================================
    // 5. FILTRAR ÚNICAMENTE REGISTROS CON DISCREPANCIA (DIFERENCIA <> 0)
    // =========================================================================
    FiltrarDiscrepancias = Table.SelectRows(AddAbsDiferencia, each [ABS_DIFERENCIA] >= 0.01),

    AddDiagnostico = Table.AddColumn(FiltrarDiscrepancias, "DIAGNÓSTICO_DISCREPANCIA", each 
        if [MET_DET] > 0 and [MET_CI] = 0 then "FALTA EN CONTROL INTERNO (Solo en Detallados)"
        else if [MET_DET] = 0 and [MET_CI] > 0 then "FALTA EN DETALLADOS (Solo en Control Interno)"
        else "DIFERENCIA EN METRAJE (Descuadre de Metros en la Guardia)", type text),

    ColsResultado = {"FECHA_FINAL", "CTR_FINAL", "MAQUINA_FINAL", "TURNO_FINAL", "SONDAJE", "MET_DET", "MET_CI", "DIFERENCIA", "DIAGNÓSTICO_DISCREPANCIA", "ID_CLAVE_UNICA_FINAL"},
    RenombrarCols = Table.RenameColumns(Table.SelectColumns(AddDiagnostico, ColsResultado), {
        {"FECHA_FINAL", "FECHA"},
        {"CTR_FINAL", "CTR"},
        {"MAQUINA_FINAL", "MAQUINA"},
        {"TURNO_FINAL", "TURNO_ESTANDAR"},
        {"MET_DET", "METRAJE_DETALLADO"},
        {"MET_CI", "METRAJE_CONTROL_INTERNO"},
        {"ID_CLAVE_UNICA_FINAL", "ID_CLAVE_UNICA"}
    }),

    SortFinal = Table.Sort(RenombrarCols, {"FECHA", "CTR", "MAQUINA"})
in
    SortFinal
